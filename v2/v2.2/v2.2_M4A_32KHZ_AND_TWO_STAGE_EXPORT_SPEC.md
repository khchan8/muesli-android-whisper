# Muesli for Android v2.2 — 32 kHz `.m4a` (AAC-LC) Audio Recording & Two-Stage Export (`Audio First, SRT Second`) Specification

**Target Repository:** `khchan8/muesli-android-whisper` (`package com.phequals7.muesli`)

## 1. Objectives for the AI Coder
1. **Record & Retain Audio in `32 kHz` `.m4a` (AAC-LC, `64 kbps`) Instead of `.wav`**:
   - Capture `AudioRecord` at `32,000 Hz` (16-bit mono PCM) in `MeetingRecorderService.kt`.
   - Stream the `32 kHz` PCM shorts directly into a hardware AAC-LC encoder (`M4aAudioEncoder.kt` using `MediaCodec` `audio/mp4a-latm` at `32000 Hz`, `64000 bps` + `MediaMuxer` `MUXER_OUTPUT_MPEG_4`) writing `<session>.m4a` (~0.48 MB/min, saving ~75% disk space vs 16 kHz WAV while preserving full 0–16 kHz wideband voice quality).
   - Simultaneously downsample `32 kHz -> 16 kHz` (`2:1` pairwise average: `((s32[2*i] + s32[2*i+1]) shr 1).toShort()`) for live Silero VAD + `SherpaOnnxEngine` chunks AND write the `16 kHz` PCM to a temporary private staging file `staging_16k.wav` in `context.cacheDir` so `SpeakerDiarizer.diarizeWavAndAlign(...)` (`Pyannote 3.0` + `CAM++`) continues working with zero model changes.
   - Delete `staging_16k.wav` immediately after `SpeakerDiarizer` finishes.

2. **Two-Stage Export to `/sdcard/Recordings` (`.m4a` Immediately on Stop, `.srt` After CAM++ Diarization)**:
   - Currently in `MeetingRecorderService.kt` (lines 358–360), `MeetingStorageManager.syncRecordingArtifacts(applicationContext, store, id)` is called only *after* `SpeakerDiarizer.diarizeWavAndAlign(...)` completes, which causes both the audio file and `.srt` to appear in `/sdcard/Recordings` at the same time at the end.
   - **Stage 1 (Immediate Audio Export on Stop)**: Immediately when `stopRecording()` finishes the recording loop and calls `m4aEncoder.stopAndRelease()`, call `MeetingStorageManager.exportAudioArtifactImmediately(applicationContext, store, id)` (and `MediaScannerConnection.scanFile` with `"audio/mp4"`) **before** starting `SpeakerDiarizer.diarizeWavAndAlign(...)`. This makes `<name>.m4a` appear in `/sdcard/Recordings` immediately when recording stops.
   - **Stage 2 (Post-Diarization `.srt` Export)**: Once `SpeakerDiarizer.diarizeWavAndAlign(...)` finishes and updates SQLite with the speaker-attributed segments, delete the temporary `staging_16k.wav` and call `MeetingStorageManager.exportSrtArtifactOnly(applicationContext, store, id)` to write and index `<name>.srt` in `/sdcard/Recordings` (and re-call `exportSrtArtifactOnly` whenever speaker names are edited in `MeetingDetailScreen.kt`).

## 2. Reference Implementation: `M4aAudioEncoder.kt`
Place at `app/src/main/java/com/phequals7/muesli/meetings/M4aAudioEncoder.kt`:

```kotlin
package com.phequals7.muesli.meetings

import android.media.MediaCodec
import android.media.MediaCodecInfo
import android.media.MediaFormat
import android.media.MediaMuxer
import android.util.Log
import java.io.File

class M4aAudioEncoder(
    private val outputFile: File,
    private val sampleRate: Int = 32000,
    private val channelCount: Int = 1,
    private val bitRate: Int = 64000
) {
    companion object {
        private const val TAG = "M4aAudioEncoder"
        private const val MIME_TYPE = MediaFormat.MIMETYPE_AUDIO_AAC
        private const val TIMEOUT_US = 10_000L
    }

    private var mediaCodec: MediaCodec? = null
    private var mediaMuxer: MediaMuxer? = null
    private var audioTrackIndex: Int = -1
    private var muxerStarted: Boolean = false
    private var totalSamplesEncoded: Long = 0L
    private val bufferInfo = MediaCodec.BufferInfo()

    init {
        outputFile.parentFile?.mkdirs()
        if (outputFile.exists()) outputFile.delete()
        val format = MediaFormat.createAudioFormat(MIME_TYPE, sampleRate, channelCount).apply {
            setInteger(MediaFormat.KEY_AAC_PROFILE, MediaCodecInfo.CodecProfileLevel.AACObjectLC)
            setInteger(MediaFormat.KEY_BIT_RATE, bitRate)
            setInteger(MediaFormat.KEY_MAX_INPUT_SIZE, 16384)
        }
        val codec = MediaCodec.createEncoderByType(MIME_TYPE)
        codec.configure(format, null, null, MediaCodec.CONFIGURE_FLAG_ENCODE)
        codec.start()
        mediaCodec = codec
        mediaMuxer = MediaMuxer(outputFile.absolutePath, MediaMuxer.OutputFormat.MUXER_OUTPUT_MPEG_4)
    }

    fun encodePcmShorts(pcm32k: ShortArray, length: Int) {
        val codec = mediaCodec ?: return
        var offset = 0
        while (offset < length) {
            val inputBufferIndex = codec.dequeueInputBuffer(TIMEOUT_US)
            if (inputBufferIndex >= 0) {
                val inputBuffer = codec.getInputBuffer(inputBufferIndex) ?: continue
                inputBuffer.clear()
                val maxShorts = (inputBuffer.remaining() / 2).coerceAtMost(length - offset)
                for (i in 0 until maxShorts) {
                    val s = pcm32k[offset + i].toInt()
                    inputBuffer.put((s and 0xFF).toByte())
                    inputBuffer.put(((s shr 8) and 0xFF).toByte())
                }
                val ptsUs = (totalSamplesEncoded * 1_000_000L) / sampleRate
                codec.queueInputBuffer(inputBufferIndex, 0, maxShorts * 2, ptsUs, 0)
                totalSamplesEncoded += maxShorts
                offset += maxShorts
            }
            drainOutput(endOfStream = false)
        }
    }

    fun stopAndRelease() {
        val codec = mediaCodec
        if (codec != null) {
            try {
                val inputBufferIndex = codec.dequeueInputBuffer(TIMEOUT_US)
                if (inputBufferIndex >= 0) {
                    val ptsUs = (totalSamplesEncoded * 1_000_000L) / sampleRate
                    codec.queueInputBuffer(inputBufferIndex, 0, 0, ptsUs, MediaCodec.BUFFER_FLAG_END_OF_STREAM)
                }
                drainOutput(endOfStream = true)
            } catch (t: Throwable) {
                Log.e(TAG, "Error flushing M4A encoder", t)
            } finally {
                runCatching { codec.stop() }
                runCatching { codec.release() }
            }
        }
        mediaCodec = null
        val muxer = mediaMuxer
        if (muxer != null) {
            try {
                if (muxerStarted) muxer.stop()
            } catch (t: Throwable) {
                Log.e(TAG, "Error stopping MediaMuxer", t)
            } finally {
                runCatching { muxer.release() }
            }
        }
        mediaMuxer = null
        muxerStarted = false
    }

    private fun drainOutput(endOfStream: Boolean) {
        val codec = mediaCodec ?: return
        val muxer = mediaMuxer ?: return
        while (true) {
            val outIndex = codec.dequeueOutputBuffer(bufferInfo, if (endOfStream) TIMEOUT_US else 0L)
            when {
                outIndex == MediaCodec.INFO_TRY_AGAIN_LATER -> if (!endOfStream) break
                outIndex == MediaCodec.INFO_OUTPUT_FORMAT_CHANGED -> {
                    if (!muxerStarted) {
                        audioTrackIndex = muxer.addTrack(codec.outputFormat)
                        muxer.start()
                        muxerStarted = true
                    }
                }
                outIndex >= 0 -> {
                    val encodedData = codec.getOutputBuffer(outIndex)
                    if (encodedData != null && bufferInfo.size > 0 && muxerStarted) {
                        if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_CODEC_CONFIG) == 0) {
                            encodedData.position(bufferInfo.offset)
                            encodedData.limit(bufferInfo.offset + bufferInfo.size)
                            muxer.writeSampleData(audioTrackIndex, encodedData, bufferInfo)
                        }
                    }
                    codec.releaseOutputBuffer(outIndex, false)
                    if ((bufferInfo.flags and MediaCodec.BUFFER_FLAG_END_OF_STREAM) != 0) break
                }
            }
        }
    }
}
```
